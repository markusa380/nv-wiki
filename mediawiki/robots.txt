# form costs no crawlable content. The /wiki duplicates cover the legacy
# prefix that Traefik strips.

Disallow: /wiki/index.php?
Disallow: /wiki/index.php/Special:
Disallow: /api.php
Disallow: /rest.php
Disallow: /wiki/api.php
Disallow: /wiki/rest.php
Disallow: /mw-config/
